1. Autoencoder – Architecture tuning


import numpy as np
import matplotlib.pyplot as plt
from tensorflow.keras.datasets import mnist
from tensorflow.keras.models import Model
from tensorflow.keras import layers
from tensorflow.keras.optimizers import Adam

e1 = [100,150,200]
e2 = [50,100,150]
d1 = [50,100,150]

(x_train,_),(x_test,_) = mnist.load_data()
x_train = x_train.astype("float32")/255.0
x_test = x_test.astype("float32")/255.0
x_train = x_train.reshape(-1,784)
x_test = x_test.reshape(-1,784)

for i in range(len(e1)):
    input_img = layers.Input(shape=(784,))
    encoded = layers.Dense(e1[i],activation="relu")(input_img)
    encoded = layers.Dense(e2[i],activation="relu")(encoded)
    decoded = layers.Dense(d1[i],activation="relu")(encoded)
    decoded = layers.Dense(784,activation="sigmoid")(decoded)

    autoencoder = Model(input_img,decoded)
    autoencoder.compile(optimizer=Adam(0.001),loss="binary_crossentropy")

    history = autoencoder.fit(x_train,x_train,epochs=10,batch_size=256,
                              shuffle=True,validation_data=(x_test,x_test),verbose=2)

    decoded_imgs = autoencoder.predict(x_test)
    plt.figure(figsize=(20,4))

    for j in range(10):
        plt.subplot(2,10,j+1)
        plt.imshow(x_test[j].reshape(28,28),cmap="gray")
        plt.axis("off")

        plt.subplot(2,10,j+11)
        plt.imshow(decoded_imgs[j].reshape(28,28),cmap="gray")
        plt.axis("off")

    plt.show()

3. Autoencoder – Loss function tuning


import numpy as np
import matplotlib.pyplot as plt
from tensorflow.keras.datasets import mnist
from tensorflow.keras.models import Model
from tensorflow.keras import layers
from tensorflow.keras.optimizers import Adam

loss = ["binary_crossentropy","mse","mae"]

(x_train,_),(x_test,_) = mnist.load_data()
x_train = x_train.astype("float32")/255.0
x_test = x_test.astype("float32")/255.0
x_train = x_train.reshape(-1,784)
x_test = x_test.reshape(-1,784)

for j in range(len(loss)):

    decoded = layers.Dense(784,activation="sigmoid")(decoded)

    autoencoder = Model(input_img,decoded)
    autoencoder.compile(optimizer=Adam(0.001),loss=loss[j])

    history = autoencoder.fit(x_train,x_train,epochs=10,batch_size=256,
