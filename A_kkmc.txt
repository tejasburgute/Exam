
####################smiles to pdbqt################

mkdir -p ligand_pdbqt

tail -n +2 compounds.csv | while IFS=, read -r ID SMILES
do
    obabel -:"$SMILES" -O ligand_pdbqt/"$ID".pdbqt --gen3d -h
done

echo "All ligands processed!"







######## vina script #########

for ligand in ligand_pdbqt/*.pdbqt
do
    name=$(basename "$ligand" .pdbqt)

    echo "Docking $name"

    vina \
    --receptor receptor.pdbqt \
    --ligand "$ligand" \
    --config conf.txt \
    --out "results/${name}_out.pdbqt" \
    --log "results/${name}_log.txt"

done



##############extract best binding affinity###############


echo "Ligand,Affinity" > docking_scores.csv

for file in results/*_log.txt
do
    name=$(basename "$file" _log.txt)
    score=$(awk '/^[[:space:]]*1[[:space:]]+-/ {print $2; exit}' "$file")
    echo "$name,$score" >> docking_scores.csv
done



echo "All docking completed!"
