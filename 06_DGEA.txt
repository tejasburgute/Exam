colnames(pdata)

# 4. QUALITY ASSESSMENT

# Boxplot
boxplot(expr, las = 2, main = "Expression Distribution")

# PCA
pca <- prcomp(t(expr), scale. = TRUE)

plot(pca$x[,1], pca$x[,2],
     xlab = "PC1",
     ylab = "PC2",
     main = "PCA Plot",
     pch = 19)

# MDS
plotMDS(expr)

# 5. SAMPLE INFORMATION
dim(pheno)
colnames(pheno)

pheno[, intersect(
    c("title", "source_name_ch1",
      "characteristics_ch1", "characteristics_ch1.1",
      "characteristics_ch1.2", "characteristics_ch1.3"),
    colnames(pheno)
), drop = FALSE]

# 6. IDENTIFY NORMAL AND TUMOR GROUPS
group <- factor(
    ifelse(grepl("tumou?r", pheno$source_name_ch1, ignore.case = TRUE),
           "Tumor",
           ifelse(grepl("normal|non.?tumou?r", pheno$source_name_ch1,
                        ignore.case = TRUE),
                  "Normal", NA)),
    levels = c("Normal", "Tumor")
)

# Check groups
table(group, useNA = "ifany")

# Keep only identified samples
keep <- !is.na(group)

expr <- expr[, keep, drop = FALSE]
pheno <- pheno[keep, , drop = FALSE]
group <- droplevels(group[keep])

# Verify
table(group)

# 7. DESIGN MATRIX
design <- model.matrix(~0 + group)

colnames(design) <- levels(group)

design
design[1:6, ]
dim(design)

# 8. CONTRAST MATRIX
contrast.matrix <- makeContrasts(
    Tumor - Normal,
    levels = design
)

contrast.matrix

# 9. FIT LINEAR MODEL
fit <- lmFit(expr, design)

# 10. APPLY CONTRAST
fit2 <- contrasts.fit(fit, contrast.matrix)

# 11. EMPIRICAL BAYES
fit2 <- eBayes(fit2)

# 12. COMPLETE RESULTS TABLE
results <- topTable(
    fit2,
    number = Inf,
    adjust.method = "BH",
    sort.by = "P"
)

head(results)

# 13. SELECT SIGNIFICANT DEGs
sig <- subset(
    results,
    adj.P.Val < 0.05 & abs(logFC) > 1
)

# UPREGULATED GENES
up <- subset(sig, logFC > 1)

# DOWNREGULATED GENES
down <- subset(sig, logFC < -1)

# DEG COUNTS
cat("Total DEGs:", nrow(sig), "\n")
cat("Upregulated:", nrow(up), "\n")
cat("Downregulated:", nrow(down), "\n")

# TOP GENES
head(up[order(-up$logFC), ], 10)
head(down[order(down$logFC), ], 10)

# 14. SAVE RESULTS
write.csv(results, "DGE_results.csv")
write.csv(sig, "Significant_DEGs.csv")
write.csv(up, "Upregulated_Genes.csv")
write.csv(down, "Downregulated_Genes.csv")

# 15. VOLCANO PLOT
EnhancedVolcano(
    results,
    lab = rownames(results),
    x = "logFC",
    y = "adj.P.Val",
    pCutoff = 0.05,
    FCcutoff = 1,
    title = "Tumor vs Normal",
    subtitle = "GSE10072",
    xlab = "Log2 Fold Change",
    ylab = "-Log10 Adjusted P-value"
)

# 16. SELECT TOP 50 GENES
topgenes <- head(
    order(results$adj.P.Val),
    50
)

# 17. HEATMAP

cat("Downregulated genes:", nrow(down), "\n")
