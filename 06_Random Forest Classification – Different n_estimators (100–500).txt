
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier

# 1. Load dataset
df = pd.read_csv(r"C:\Users\TEJAS\OneDrive\Desktop\Project Kushagra Sir\alzheimers_disease_data.csv")

X = df.iloc[:, 0:32]
y = df.iloc[:, 32]

# 2. Split data
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.20, random_state=42
)

y_train = np.array(y_train)
y_test = np.array(y_test)

# 3. Initialize lists
accuracy = []
precision = []
recall = []

accuracy_t = []
precision_t = []
recall_t = []

ne = [100, 200, 300, 400, 500]

# 4. Train models
for i in range(0, len(ne)):

    clf = RandomForestClassifier(
        n_estimators=ne[i], criterion='gini', random_state=42
    )

    clf.fit(X_train, y_train)

    y_pred_train = clf.predict(X_train)
    y_pred_test = clf.predict(X_test)

    # Training metrics
    TP = TN = FP = FN = 0

    for j in range(len(y_pred_train)):
        if y_pred_train[j] == 1 and y_train[j] == 1: TP += 1
        elif y_pred_train[j] == 0 and y_train[j] == 0: TN += 1
        elif y_pred_train[j] == 1 and y_train[j] == 0: FP += 1
        elif y_pred_train[j] == 0 and y_train[j] == 1: FN += 1

    acc = (TP + TN) / (TP + TN + FP + FN)
    prec = TP / (TP + FP) if (TP + FP) != 0 else 0
    rec = TP / (TP + FN) if (TP + FN) != 0 else 0

    accuracy.append(acc)
    precision.append(prec)
    recall.append(rec)

    # Testing metrics
    tp = tn = fp = fn = 0

    for k in range(len(y_pred_test)):
        if y_pred_test[k] == 1 and y_test[k] == 1: tp += 1
        elif y_pred_test[k] == 0 and y_test[k] == 0: tn += 1
        elif y_pred_test[k] == 1 and y_test[k] == 0: fp += 1
        elif y_pred_test[k] == 0 and y_test[k] == 1: fn += 1

    acc_t = (tp + tn) / (tp + tn + fp + fn)
    prec_t = tp / (tp + fp) if (tp + fp) != 0 else 0
    rec_t = tp / (tp + fn) if (tp + fn) != 0 else 0

    accuracy_t.append(acc_t)
    precision_t.append(prec_t)
    recall_t.append(rec_t)

# 5. Print results
print("Training Accuracy:", accuracy)
print("Training Precision:", precision)
print("Training Recall:", recall)

print("Testing Accuracy:", accuracy_t)
print("Testing Precision:", precision_t)
print("Testing Recall:", recall_t)

# 6. Plot graphs
models = ["100 Trees", "200 Trees", "300 Trees", "400 Trees", "500 Trees"]

for i in range(5):

    metrics = ["Accuracy", "Precision", "Recall"]

    train_values = [accuracy[i], precision[i], recall[i]]
    test_values = [accuracy_t[i], precision_t[i], recall_t[i]]

    x = np.arange(len(metrics))
    width = 0.35

    fig, ax = plt.subplots(figsize=(8, 5))

    train_bar = ax.bar(x - width/2, train_values, width, label="Train")
    test_bar = ax.bar(x + width/2, test_values, width, label="Test")

    ax.set_xlabel("Metrics")
    ax.set_ylabel("Score")
    ax.set_title(models[i])
    ax.set_xticks(x)
    ax.set_xticklabels(metrics)
    ax.set_ylim(0, 1.1)

    ax.bar_label(train_bar, fmt="%.3f", padding=3)
    ax.bar_label(test_bar, fmt="%.3f", padding=3)

    ax.legend()
    plt.tight_layout()
    plt.show()
